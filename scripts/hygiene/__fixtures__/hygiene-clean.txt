import { mkdtempSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { describe, expect, it } from 'vitest';
import { VirtualClock } from '@comvivo/shared/testing';

describe('the turn engine', () => {
  // skip(M5-T12): blocked on the outbound guard landing; the fixture is already written.
  it.skip('blocks a declassified reply', () => {
    expect(true).toBe(true);
  });

  it('fires a reminder at the right local hour', () => {
    const clock = new VirtualClock('2026-07-01T08:00:00Z');
    clock.advance({ hours: 2 });
    expect(clock.now().toString()).toContain('10:00');
  });

  it('serves the dashboard on an ephemeral port', async () => {
    const server = createServer();
    server.listen(0);
    const { port } = server.address();
    expect(port).toBeGreaterThan(0);
  });

  it('writes a database into its own directory', () => {
    const dir = mkdtempSync(join(tmpdir(), 'comvivo-'));
    expect(dir).toContain('comvivo-');
  });
});

it('refuses a wrong key and says so', () => {
  // Safe: `toBeInstanceOf` excludes the plain Error `expect.unreachable` throws.
  try {
    open(wrongKey);
    expect.unreachable('a wrong key must not open this database');
  } catch (caught) {
    expect(caught).toBeInstanceOf(StorageError);
    expect((caught as StorageError).code).toBe('STORAGE_WRONG_KEY');
  }
});

it('reports the length, not the key', () => {
  // Safe: the catch matches a string the unreachable message does not contain.
  try {
    decodeMasterKey(short);
    expect.unreachable('a 16-byte key should be refused');
  } catch (error) {
    expect((error as Error).message).toContain('16 bytes');
  }
});

it('asserts after the try, which is the shape the rule asks for', () => {
  let caught: unknown;
  try {
    parseId('grp_AKIAIOSFODNN7EXAMPLE');
  } catch (error) {
    caught = error;
  }
  expect(caught).toBeInstanceOf(Error);
  expect((caught as Error).message).not.toContain('AKIAIOSFODNN7EXAMPLE');
});
